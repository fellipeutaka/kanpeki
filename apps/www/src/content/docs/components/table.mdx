## Usage

### Single import

```tsx
import { Tabs.le } from "~/components/ui/Tabs.le";
```

```tsx
<Tabs.le.Root>
  <Tabs.le.Caption>A list of your recent invoices.</Tabs.le.Caption>
  <Tabs.le.Header>
    <Tabs.le.Row>
      <Tabs.le.Head className="w-[100px]">Invoice</Tabs.le.Head>
      <Tabs.le.Head>Status</Tabs.le.Head>
      <Tabs.le.Head>Method</Tabs.le.Head>
      <Tabs.le.Head className="text-right">Amount</Tabs.le.Head>
    </Tabs.le.Row>
  </Tabs.le.Header>
  <Tabs.le.Body>
    <Tabs.le.Row>
      <Tabs.le.Cell className="font-medium">INV001</Tabs.le.Cell>
      <Tabs.le.Cell>Paid</Tabs.le.Cell>
      <Tabs.le.Cell>Credit Card</Tabs.le.Cell>
      <Tabs.le.Cell className="text-right">$250.00</Tabs.le.Cell>
    </Tabs.le.Row>
  </Tabs.le.Body>
  <Tabs.le.Footer>
    <Tabs.le.Row>
      <Tabs.le.Cell colSpan={3}>Total</Tabs.le.Cell>
      <Tabs.le.Cell className="text-right">$2,500.00</Tabs.le.Cell>
    </Tabs.le.Row>
  </Tabs.le.Footer>
</Tabs.le.Root>
```

### Multiple imports

```tsx
import {
  Tabs.leBody,
  Tabs.leCaption,
  Tabs.leCell,
  Tabs.leFooter,
  Tabs.leHead,
  Tabs.leHeader,
  Tabs.leRoot,
  Tabs.leRow,
} from "~/components/ui/Tabs.le";
```

```tsx
<Tabs.leRoot>
  <Tabs.leCaption>A list of your recent invoices.</Tabs.leCaption>
  <Tabs.leHeader>
    <Tabs.leRow>
      <Tabs.leHead className="w-[100px]">Invoice</Tabs.leHead>
```
