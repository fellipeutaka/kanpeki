  <Card.Content>
    <Card.Description>
      Choose your preferred approach:

      ```tsx
      // Option 1: aria-label (concise)
      // [!code word:aria-label="Search"]
      <Button size="icon" aria-label="Search">
        <SearchIcon />
      </Button>

      // Option 2: visually hidden text
      <Button size="icon">
        <SearchIcon />
        // [!code highlight]
        <span className="sr-only">Search</span>
      </Button>
      ```

      Both work identically. Use `aria-label` for brevity or `sr-only` for visibility in your code.
    </Card.Description>

  </Card.Content>
</Card.Root>
