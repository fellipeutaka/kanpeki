<Tabs.Content id="single">

```tsx
import { DateField } from "~/components/ui/date-field";

<DateField.Root>
  <Label />
  <DateField.Input>
    {(segment) => <DateField.Segment segment={segment} />}
  </DateField.Input>
</DateField.Root>;
```
