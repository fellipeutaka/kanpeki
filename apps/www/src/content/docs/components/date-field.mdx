## Usage

### Single import

```tsx
import { DateField } from "~/components/ui/date-field";
```

```tsx
<DateField.Root>
  <Label />
  <DateField.Input>
    {(segment) => <DateField.Segment segment={segment} />}
  </DateField.Input>
</DateField.Root>
```

### Multiple imports

```tsx
import {
  DateFieldInput,
  DateFieldRoot,
  DateFieldSegment,
} from "~/components/ui/date-field";
```

```tsx
<DateFieldRoot>
  <Label />
  <DateFieldInput>
    {(segment) => <DateFieldSegment segment={segment} />}
  </DateFieldInput>
</DateFieldRoot>
```
