## Usage

### Single import

```tsx
import { TextField } from "~/components/ui/text-field";
```

```tsx
<TextField.Provider>
  <TextField.Root>
    <TextField.Slot />
    <TextField.Input />
    <TextField.Slot />
  </TextField.Root>
</TextField.Provider>
```

### Multiple imports

```tsx
import {
  TextFieldProvider,
  TextFieldRoot,
  TextFieldSlot,
  TextFieldInput,
} from "~/components/ui/text-field";
```

```tsx
<TextFieldProvider>
  <TextFieldRoot>
    <TextFieldSlot />
    <TextFieldInput />
    <TextFieldSlot />
  </TextFieldRoot>
</TextFieldProvider>
```
