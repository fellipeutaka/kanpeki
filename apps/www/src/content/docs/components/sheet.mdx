## Anatomy

<Tabs.Root>

<Tabs.List>
  <Tabs.Trigger id="single">Single import</Tabs.Trigger>
  <Tabs.Trigger id="multiple">Multiple imports</Tabs.Trigger>
</Tabs.List>
<Tabs.Content id="single">

```tsx
import { Button } from "~/components/ui/button";
import { Sheet } from "~/components/ui/sheet";

<Sheet.Root>
  <Button>Trigger</Button>

  <Sheet.Overlay>
    <Sheet.Modal>
      <Sheet.Content>
        <Sheet.Header>
          <Sheet.Title />
          <Sheet.Description />
        </Sheet.Header>

        <Sheet.Footer>
          <Button slot="close" />
        </Sheet.Footer>

        <Sheet.Close />
      </Sheet.Content>
    </Sheet.Modal>
  </Sheet.Overlay>
</Sheet.Root>;
```

</Tabs.Content>

<Tabs.Content id="multiple">

```tsx
import { Button } from "~/components/ui/button";
import {
  SheetClose,
  SheetContent,
  SheetDescription,
  SheetFooter,
  SheetHeader,
  SheetModal,
  SheetOverlay,
  SheetRoot,
  SheetTitle,
} from "~/components/ui/sheet";

<SheetRoot>
  <Button>Trigger</Button>

  <SheetOverlay>
    <SheetModal>
      <SheetContent>
        <SheetHeader>
          <SheetTitle />
          <SheetDescription />
        </SheetHeader>

        <SheetFooter>
          <Button slot="close" />
        </SheetFooter>

        <SheetClose />
      </SheetContent>
    </SheetModal>
  </SheetOverlay>
</SheetRoot>;
```

</Tabs.Content>

</Tabs.Root>

## Examples

### Side

Use the `side{:tsx}` prop on `<Sheet.Modal>{:tsx}` to indicate the edge of the screen where the component will appear. The values can be `top{:tsx}`, `right{:tsx}`, `bottom{:tsx}`, or `left{:tsx}`.
