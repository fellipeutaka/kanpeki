## Usage

### Single import

```tsx
import { Button } from "~/components/ui/button";
import { Tooltip } from "~/components/ui/tooltip";
```

```tsx
<Tooltip.Root>
  <Button></Button>

  <Tooltip.Content>
    <p></p>

    <Tooltip.Arrow />
  </Tooltip.Content>
</Tooltip.Root>
```

### Multiple imports

```tsx
import { Button } from "~/components/ui/button";
import {
  TooltipArrow,
  TooltipContent,
  TooltipRoot,
} from "~/components/ui/tooltip";
```

```tsx
<TooltipRoot>
  <Button></Button>
  <TooltipContent>
    <p></p>

    <TooltipArrow />
  </TooltipContent>
</TooltipRoot>
```
