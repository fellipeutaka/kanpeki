## Usage

### Single import

```tsx
import { Slider } from "~/components/ui/slider";
```

```tsx
<Slider.Root>
  <Slider.Track>
    <Slider.Filler />
    <Slider.Thumb />
  </Slider.Track>
</Slider.Root>
```

### Multiple imports

```tsx
import {
  SliderFiller,
  SliderRoot,
  SliderThumb,
  SliderTrack,
} from "~/components/ui/slider";
```

```tsx
<SliderRoot>
  <SliderTrack>
    <SliderFiller />
    <SliderThumb />
  </SliderTrack>
</SliderRoot>
```
