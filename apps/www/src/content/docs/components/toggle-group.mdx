## Anatomy

```tsx
import { ToggleGroup } from "~/components/ui/toggle-group";

<ToggleGroup.Root selectionMode="single">
  <ToggleGroup.Item id="a">A</ToggleGroup.Item>
  <ToggleGroup.Item id="b">B</ToggleGroup.Item>
  <ToggleGroup.Item id="c">C</ToggleGroup.Item>
</ToggleGroup.Root>;
```
