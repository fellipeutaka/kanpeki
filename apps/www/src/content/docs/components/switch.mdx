## Usage

### Single import

```tsx
import { Switch } from "~/registry/ui/switch";
```

```tsx
<Switch.Root>
  <Switch.Track>
    <Switch.Thumb />
  </Switch.Track>
</Switch.Root>
```

### Multiple imports

```tsx
import { SwitchRoot, SwitchTrack, SwitchThumb } from "~/registry/ui/switch";
```

```tsx
<SwitchRoot>
  <SwitchTrack>
    <SwitchThumb />
  </SwitchTrack>
</SwitchRoot>
```
