<Tabs.Content id="single">

```tsx
import { Empty } from "~/components/ui/empty";

<Empty.Root>
  <Empty.Header>
    <Empty.Media variant="icon">
      <Icon />
    </Empty.Media>
    <Empty.Title>No data</Empty.Title>
    <Empty.Description>No data found</Empty.Description>
  </Empty.Header>
  <Empty.Content>
    <Button>Add data</Button>
  </Empty.Content>
</Empty.Root>;
```

import {
