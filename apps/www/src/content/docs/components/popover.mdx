## Usage

### Single import

```tsx
import { Button } from "~/components/ui/button";
import { Popover } from "~/components/ui/popover";
```

```tsx
<Popover.Root>
  <Button>...</Button>

  <Popover.Content>
    <Popover.Arrow />
    ...
  </Popover.Content>
</Popover.Root>
```

### Multiple imports

```tsx
import { Button } from "~/components/ui/button";
import {
  PopoverArrow,
  PopoverContent,
  PopoverRoot,
} from "~/components/ui/popover";
```

```tsx
<PopoverRoot>
  <Button>...</Button>
  <PopoverContent>
    <PopoverArrow />
    ...
  </PopoverContent>
</PopoverRoot>
```
