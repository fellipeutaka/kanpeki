## Installation

The Select is built using a composition of the `<Popover />{:tsx}` and the `<Listbox />{:tsx}` components.

See installation instructions for the [Popover](/docs/components/popover#installation) and the [Listbox](/docs/components/list-box#installation) components.

```tsx
import { Listbox } from "~/components/ui/list-box";
import { Popover } from "~/components/ui/popover";
import { Select } from "~/components/ui/select";

<Select.Root>
  <Select.Trigger>
    <Select.Value />
  </Select.Trigger>
  <Popover.Content>
    <Listbox.Root>
      <Listbox.Item></Listbox.Item>

import { ListboxItem, ListboxRoot } from "~/components/ui/list-box";
import { PopoverContent } from "~/components/ui/popover";
import { SelectRoot, SelectTrigger, SelectValue } from "~/components/ui/select";

<SelectRoot>
  <SelectTrigger>
    <SelectValue />
  </SelectTrigger>
  <PopoverContent>
    <ListboxRoot>
      <ListboxItem />
    </ListboxRoot>
  </PopoverContent>
</SelectRoot>;
```
