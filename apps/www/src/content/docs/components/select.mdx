## Usage

### Single import

```tsx
import { Select } from "~/components/ui/select";
```

```tsx
<Select.Root>
  <Select.Trigger />

  <Select.Popover>
    <Select.Content items={fruits}>
      {(item) => (
        <Select.Item id={item.id} textValue={item.name}>
          {item.name}
          <Icons.Check className="size-0 opacity-0 group-selected:size-4 group-selected:opacity-100" />
        </Select.Item>
      )}
    </Select.Content>
  </Select.Popover>
</Select.Root>
```

### Multiple imports

```tsx
import {
  SelectContent,
  SelectItem,
  SelectPopover,
  SelectRoot,
  SelectTrigger,
} from "~/components/ui/select";
```

```tsx
<SelectRoot>
  <SelectTrigger />

  <SelectPopover>
    <SelectContent items={fruits}>
      {(item) => (
        <SelectItem id={item.id} textValue={item.name}>
          {item.name}
          <Icons.Check className="size-0 opacity-0 group-selected:size-4 group-selected:opacity-100" />
        </SelectItem>
      )}
    </SelectContent>
  </SelectPopover>
</SelectRoot>
```
