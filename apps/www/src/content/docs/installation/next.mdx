<Steps>

### Create project

Start by creating a new Next.js project using `create-next-app`:

```bash
npx create-next-app@latest --tailwind --ts --empty --app --src-dir --turbopack
```

You will be asked a few questions to create the project:

```bash
? What is your project named? » my-app
? Which linter would you like to use? » Biome
? Would you like to customize the default import alias (@/*)? » No / Yes
```

Currently, I prefer to use [Biome.js](https://biomejs.dev) instead of ESLint. You can use ESLint if you prefer.
Also I prefer to use `~` as the alias. You can use `@` or any other alias if you prefer.

### Run the shadcn CLI

Run the `shadcn` init command to setup your project:

```bash
npx shadcn@latest init
```

### Fonts

I use [Geist](https://vercel.com/font) as the default font.
You can use any font you like.

Here's how I configure Geist for Next.js:

**1. Create a `font.ts` file inside `src/config`:**

```ts title=font.ts
import { Geist } from "next/font/google";

const sans = Geist({
  subsets: ["latin"],
  variable: "--font-sans",
});

export const fonts = [sans.variable];
```

**2. Import the font in the root layout:**

```tsx title=app/layout.tsx
import "~/styles/globals.css";

import type { Metadata, Viewport } from "next";
import { fonts } from "~/config/fonts"; // [!code highlight]
import { cx } from "cva";

export const metadata: Metadata = {
  title: {
    default: "Next.js app",
    template: "%s | Next.js app",
  },
};

export const viewport: Viewport = {
  themeColor: [
    { media: "(prefers-color-scheme: light)", color: "white" },
    { media: "(prefers-color-scheme: dark)", color: "black" },
  ],
};

interface LayoutProps {
  children: React.ReactNode;
}

export default function Layout({ children }: LayoutProps) {
  return (
    // [!code word:cx(fonts)]
    <html className={cx(fonts)} lang="en" suppressHydrationWarning>
      <body>
        {children}
      </body>
    </html>
  );
}
```

**3. Configure `--font-sans` in `globals.css`**

```ts title="globals.css"
@import "tailwindcss";

@plugin "tailwindcss-react-aria-components";
@plugin "tailwindcss-motion";

@custom-variant dark (&:where(.dark, .dark *));

@theme inline {
  // [!code word:var(--font-sans)]
  --font-sans: var(--font-sans), ui-sans-serif, system-ui, sans-serif,
    "Apple Color Emoji", "Segoe UI Emoji", "Segoe UI Symbol", "Noto Color Emoji";

  ...
}
```

### App structure

Here's how I structure my Next.js apps. You can use this as a reference:

<Files className="mt-6">
  <Folder name="src">
    <Folder name="app">
      <File name="layout.tsx" />
      <File name="page.tsx" />
      <File name="providers.tsx" />
    </Folder>
    <Folder name="components">
      <Folder name="ui">
        <Folder name="button">
          <File name="button.tsx" />
          <File name="index.ts" />
          <File name="styles.ts" />
        </Folder>
        <Folder name="dialog">
          <File name="dialog.tsx" />
          <File name="index.ts" />
          <File name="namespace.ts" />
          <File name="styles.ts" />
        </Folder>
        <Folder name="Tabs.s">
          <File name="index.ts" />
          <File name="namespace.ts" />
          <File name="styles.ts" />
          <File name="Tabs.s.tsx" />
        </Folder>
      </Folder>
    </Folder>
    <Folder name="config">
      <File name="fonts.ts" />
    </Folder>
    <Folder name="lib">
      <File name="cva.ts" />
    </Folder>
    <Folder name="styles">
      <File name="globals.css" />
    </Folder>
  </Folder>
  <File name="components.json" />
  <File name="next.config.ts" />
  <File name="package.json" />
  <File name="postcss.config.js" />
  <File name="tsconfig.json" />
</Files>

- I place the UI components in the `src/components/ui` folder.
- The `lib` folder contains configuration for external libraries.
- The `styles` folder contains the global CSS.

### That's it ✨

You can now start adding components to your project.

```bash
npx shadcn@latest add @kanpeki/button
```

The command above will add the `Button` component to your project. You can then import it like this:

```tsx
import { Button } from "~/components/ui/button"; // [!code highlight]

export default function Home() {
  return (
    <div>
      {/* [!code highlight] */}
      <Button>Click me</Button>
    </div>
  );
}
```
